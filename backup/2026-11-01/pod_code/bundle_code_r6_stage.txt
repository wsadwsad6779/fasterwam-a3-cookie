'''Phase 1 step B: stream-sha256 every target listed by backup_scan.py.

Runs in the background: the three merged checkpoints are 11 GB each and dominate
the wall time. One JSONL line per finished file, flushed immediately, so progress
is pollable and a restart skips what is already hashed.
'''
import hashlib
import json
import time
from pathlib import Path

OUT = Path('/workspace/_r6_stage/out')
LIST = OUT / 'backup_filelist.json'
LOG = OUT / 'backup_hashes.jsonl'
CHUNK = 8 * 1024 * 1024


def done_paths():
    seen = set()
    if LOG.exists():
        for line in LOG.read_text().splitlines():
            try:
                seen.add(json.loads(line)['path'])
            except Exception:
                continue
    return seen


def main():
    entries = json.loads(LIST.read_text())['entries']
    entries.sort(key=lambda e: e['bytes'])
    have = done_paths()
    with LOG.open('a') as fh:
        for i, e in enumerate(entries):
            if e['path'] in have:
                continue
            t0 = time.time()
            h = hashlib.sha256()
            n = 0
            try:
                with open(e['path'], 'rb') as f:
                    while True:
                        b = f.read(CHUNK)
                        if not b:
                            break
                        h.update(b)
                        n += len(b)
            except Exception as ex:
                rec = {'path': e['path'], 'error': type(ex).__name__ + ' ' + str(ex)[:120]}
            else:
                rec = {'path': e['path'], 'sha256': h.hexdigest(), 'read_bytes': n,
                       'seconds': round(time.time() - t0, 2),
                       'utc': time.strftime('%Y-%m-%dT%H:%M:%SZ', time.gmtime())}
                if n != e['bytes']:
                    rec['size_drift'] = [e['bytes'], n]
            fh.write(json.dumps(rec) + '\n')
            fh.flush()
            print('%d/%d %s %s' % (i + 1, len(entries), rec.get('sha256', rec.get('error'))[:16],
                                   e['path'].replace('/workspace/', '')), flush=True)
    print('HASH_DONE lines=%d' % len(done_paths()), flush=True)


main()

"""Independent test branch for light-state completeness.

Nothing here is imported by training or evaluation, and no r1..r5 checkpoint is
read for writing: the live module stays at
`/workspace/FasterWAM/src/fasterwam/light_state.py` and this file only wraps it.

Why it exists: `lora_named_parameters()` keeps only names containing "lora_",
but the trainer also optimises `proprio_encoder` (`trainer.py:86-88`), so every
light state silently drops 2 of the 786 trainable tensors while `optimizer.pt`
still carries their Adam moments. A v2 adapter carries every trainable tensor
and refuses to pretend a legacy file is complete.
"""
from __future__ import annotations

import json
import os
from pathlib import Path

import torch

from fasterwam import light_state as L

ADAPTER_FORMAT_V2 = "fasterwam-lora-adapter-v2"
STATE_FORMAT_V2 = "fasterwam-lora-light-state-v2"


def trainable_named_parameters(model):
    return [(n, p) for n, p in model.named_parameters() if p.requires_grad]


def coverage(model):
    """What v1 keeps, what v1 drops, and what the optimiser is actually stepping."""
    trainable = trainable_named_parameters(model)
    lora = [(n, p) for n, p in trainable if "lora_" in n]
    extra = [(n, p) for n, p in trainable if "lora_" not in n]
    return {
        "trainable_tensors": len(trainable),
        "trainable_elements": sum(int(p.numel()) for _, p in trainable),
        "kept_by_v1": len(lora),
        "kept_by_v1_elements": sum(int(p.numel()) for _, p in lora),
        "dropped_by_v1": [{"name": n, "shape": list(p.shape), "elements": int(p.numel())}
                          for n, p in extra],
        "dropped_by_v1_elements": sum(int(p.numel()) for _, p in extra),
    }


def _split_trainable(model):
    lora, extra = {}, {}
    for name, param in trainable_named_parameters(model):
        held = param.detach().to("cpu", copy=True)
        (lora if "lora_" in name else extra)[name] = held
    return lora, extra


def save_adapter_v2(model, path, meta=None):
    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)
    lora, extra = _split_trainable(model)
    tmp = path.with_name(path.name + ".tmp")
    torch.save({"format": ADAPTER_FORMAT_V2,
                "lora": lora,
                "extra_trainable": extra,
                "meta": dict(meta or {})}, tmp)
    os.replace(tmp, path)
    return path


def _copy_into(named, blob, path):
    unknown = sorted(set(blob) - set(named))
    if unknown:
        raise RuntimeError("%s: adapter has tensors the model does not: %s"
                           % (path, unknown[:5]))
    shapes = ["%s: file%s model%s" % (n, list(blob[n].shape), list(named[n].shape))
              for n in blob if list(blob[n].shape) != list(named[n].shape)]
    if shapes:
        raise RuntimeError("%s: shape mismatch: %s" % (path, shapes[:5]))
    with torch.no_grad():
        for name, value in blob.items():
            param = named[name]
            param.copy_(value.to(device=param.device, dtype=param.dtype))


def load_adapter_v2(model, path, map_location="cpu", allow_legacy=False):
    """Restore every trainable tensor. A v1 file is incomplete and says so.

    Matching runs against `model.named_parameters()` rather than the current
    trainable set, so a model built for inference (where `requires_grad` may be
    switched off) still loads; the completeness check is the other direction --
    anything trainable now but absent from the file is reported.
    """
    path = Path(path)
    payload = torch.load(path, map_location=map_location, weights_only=False)
    fmt = payload.get("format")
    named = dict(model.named_parameters())
    trainable_names = {n for n, _ in trainable_named_parameters(model)}

    if fmt == L.ADAPTER_FORMAT:
        blob = dict(payload["lora"])
        missing = sorted(trainable_names - set(blob))
        if missing and not allow_legacy:
            raise RuntimeError(
                "%s: v1 adapter covers %d of %d trainable tensors. These were "
                "trainable but never saved: %s. The file cannot represent the "
                "trained model; it only restores the LoRA part and leaves the "
                "rest at whatever the base checkpoint supplied. Pass "
                "allow_legacy=True to load it anyway."
                % (path, len(blob), len(trainable_names), missing))
        _copy_into(named, blob, path)
        return {"format": "v1", "restored_tensors": len(blob),
                "trainable_tensors": len(trainable_names),
                "not_restored": missing, "meta": payload.get("meta", {})}

    if fmt != ADAPTER_FORMAT_V2:
        raise ValueError("%s: not a recognised adapter checkpoint (%s)" % (path, fmt))

    blob = dict(payload["lora"])
    blob.update(payload.get("extra_trainable", {}))
    missing = sorted(trainable_names - set(blob))
    if missing:
        raise RuntimeError("%s: v2 adapter is missing trainable tensors %s"
                           % (path, missing[:5]))
    _copy_into(named, blob, path)
    return {"format": "v2", "restored_tensors": len(blob),
            "trainable_tensors": len(trainable_names),
            "not_restored": [], "meta": payload.get("meta", {})}


def save_light_state_v2(state_dir, *, model, optimizer, scheduler, sampler,
                        global_step, epoch, batch_in_epoch, extra=None,
                        adapter_meta=None):
    """Same discipline as `save_light_state`: tmp+replace, state file last."""
    state_dir = Path(state_dir)
    state_dir.mkdir(parents=True, exist_ok=True)

    def _replace(tmp_name, obj):
        tmp = state_dir / (tmp_name + ".tmp")
        torch.save(obj, tmp)
        os.replace(tmp, state_dir / tmp_name)

    cov = coverage(model)
    save_adapter_v2(model, state_dir / "adapter.pt",
                    meta={"global_step": int(global_step),
                          "trainable_tensors": cov["trainable_tensors"],
                          "dropped_by_v1": [d["name"] for d in cov["dropped_by_v1"]],
                          **(adapter_meta or {})})
    _replace("optimizer.pt", L._optimizer_cpu_state(optimizer))
    _replace("scheduler.pt", getattr(scheduler, "state_dict", lambda: {})())
    _replace("rng.pt", L.capture_rng())

    state = {"format": STATE_FORMAT_V2,
             "global_step": int(global_step),
             "epoch": int(epoch),
             "batch_in_epoch": int(batch_in_epoch),
             "sampler": L._sampler_state(sampler),
             "extra": dict(extra or {}),
             "coverage": cov,
             "files": {"adapter": "adapter.pt", "optimizer": "optimizer.pt",
                       "scheduler": "scheduler.pt", "rng": "rng.pt"}}
    tmp = state_dir / "trainer_state.json.tmp"
    tmp.write_text(json.dumps(state, indent=2))
    os.replace(tmp, state_dir / "trainer_state.json")
    return state


def load_light_state_v2(state_dir, *, model, optimizer, scheduler, sampler,
                        map_location="cpu", allow_legacy=False):
    state_dir = Path(state_dir)
    state_file = state_dir / "trainer_state.json"
    if not state_file.exists():
        raise FileNotFoundError("%s missing; refusing a partial restore" % state_file)
    state = json.loads(state_file.read_text())
    if state.get("format") != STATE_FORMAT_V2:
        raise ValueError("%s: unknown light-state format %s"
                         % (state_dir, state.get("format")))

    report = load_adapter_v2(model, state_dir / "adapter.pt",
                             map_location=map_location, allow_legacy=allow_legacy)

    inner = getattr(optimizer, "optimizer", optimizer)
    inner.load_state_dict(torch.load(state_dir / "optimizer.pt",
                                     map_location=map_location, weights_only=False))
    device = next(p.device for group in inner.param_groups for p in group["params"])
    L._optimizer_state_to_device(optimizer, device)

    scheduler_state = torch.load(state_dir / "scheduler.pt",
                                 map_location=map_location, weights_only=False)
    if scheduler_state:
        scheduler.load_state_dict(scheduler_state)

    L.restore_rng(torch.load(state_dir / "rng.pt", map_location="cpu", weights_only=False))
    L._apply_sampler_state(sampler, state["sampler"])
    state["adapter_report"] = report
    return state

'''v3 light state: persist exactly the tensors the optimiser owns.

v1 keyed the save set on the name pattern lora_ and silently dropped
proprio_encoder, which trainer.py:88 puts in the AdamW list. A v2 draft keyed on
requires_grad instead; on the live model that is 1637 tensors / 8,710,770,916
elements, because trainer.py:288-294 freezes by pattern without clearing the
flag on everything else -- writing it would have cost ~9 GB per save and filled
/workspace. Optimiser membership is the only set that means that training changed
this, so v3 selects by parameter identity and refuses to write when the
projected size exceeds the caller's budget.
'''
import json
import os
import shutil
from pathlib import Path

import torch

ADAPTER_FORMAT_V3 = 'fasterwam-lora-adapter-v3'
STATE_FORMAT_V3 = 'fasterwam-lora-light-state-v3'
V1_ADAPTER_FORMAT = 'fasterwam-lora-adapter-v1'

_WIDTH = {torch.bfloat16: 2, torch.float16: 2, torch.float32: 4, torch.float64: 8}


def _inner(optimizer):
    return getattr(optimizer, 'optimizer', optimizer)


def optimizer_membership(optimizer):
    inner = _inner(optimizer)
    members = [p for g in inner.param_groups for p in g['params']]
    return members


def select_saved(model, optimizer):
    '''Name -> tensor for every parameter the optimiser can modify.

    Identity first, because that is the truth. If the optimiser holds tensors
    that are not the model's own parameters (a sharding wrapper would), fall back
    to the trainer.py:85-88 construction order and accept it only when the
    shape/dtype sequence matches exactly.
    '''
    named = list(model.named_parameters())
    members = optimizer_membership(optimizer)
    ids = {id(p): n for n, p in named}
    hits = [(n, p) for p in members if id(p) in ids for n in [ids[id(p)]]]
    method = 'identity'
    if len(hits) != len(members):
        cand = [p for p in model.dit.parameters() if p.requires_grad]
        proprio = getattr(model, 'proprio_encoder', None)
        if proprio is not None:
            cand += [p for p in proprio.parameters() if p.requires_grad]
        cand_names = []
        for n, p in named:
            if id(p) in {id(c) for c in cand}:
                cand_names.append((n, p))
        sig = [(tuple(p.shape), str(p.dtype)) for p in cand]
        msig = [(tuple(p.shape), str(p.dtype)) for p in members]
        if sig != msig:
            raise RuntimeError('optimizer membership unresolved: identity matched %d of %d '
                               'and the trainer-order fallback does not reproduce it '
                               '(shapes %s vs %s). Refusing to guess what to save.'
                               % (len(hits), len(members), msig[:4], sig[:4]))
        hits, method = cand_names, 'positional_verified'
    seen = set()
    out = []
    for n, p in hits:
        if n in seen:
            continue
        seen.add(n)
        out.append((n, p))
    return {'selected': out, 'method': method, 'optimizer_members': len(members),
            'selected_tensors': len(out)}


def projected_bytes(selected):
    return sum(int(p.numel()) * _WIDTH.get(p.dtype, 4) for _, p in selected)


def free_bytes(path):
    d = Path(path)
    while not d.exists() and d != d.parent:
        d = d.parent
    return int(shutil.disk_usage(str(d)).free)


def coverage(model, optimizer):
    '''Three different sets, kept apart on purpose.'''
    named = list(model.named_parameters())
    sel = select_saved(model, optimizer)
    sel_names = {n for n, _ in sel['selected']}
    req = [n for n, p in named if p.requires_grad]
    lora = [n for n in req if 'lora_' in n]
    return {'named_tensors': len(named),
            'named_elements': sum(int(p.numel()) for _, p in named),
            'requires_grad_tensors': len(req),
            'requires_grad_elements': sum(int(p.numel()) for n, p in named if n in set(req)),
            'optimizer_members': sel['optimizer_members'],
            'saved_tensors': len(sel_names),
            'saved_elements': sum(int(p.numel()) for n, p in named if n in sel_names),
            'saved_lora_tensors': len([n for n in sel_names if 'lora_' in n]),
            'saved_nonlora_names': sorted(n for n in sel_names if 'lora_' not in n),
            'requires_grad_but_not_optimised': len([n for n in req if n not in sel_names]),
            'requires_grad_but_not_optimised_elements': sum(
                int(p.numel()) for n, p in named if p.requires_grad and n not in sel_names),
            'kept_by_v1': len(lora),
            'kept_by_v1_elements': sum(int(p.numel()) for n, p in named if n in set(lora)),
            'dropped_by_v1': sorted(n for n, p in named
                                    if p.requires_grad and 'lora_' not in n
                                    and n in sel_names),
            'dropped_by_v1_elements': sum(int(p.numel()) for n, p in named
                                          if p.requires_grad and 'lora_' not in n
                                          and n in sel_names),
            'selection_method': sel['method'],
            'projected_adapter_bytes': projected_bytes(sel['selected'])}


def save_adapter_v3(model, optimizer, path, meta=None, max_bytes=None):
    path = Path(path)
    cov = coverage(model, optimizer)
    sel = select_saved(model, optimizer)['selected']
    est = projected_bytes(sel)
    if max_bytes is None:
        max_bytes = int(free_bytes(path.parent) * 0.25)
    if est > max_bytes:
        raise RuntimeError('%s: v3 adapter would write %d B but the budget is %d B; '
                           'nothing was written.' % (path, est, max_bytes))
    payload = {'format': ADAPTER_FORMAT_V3,
               'lora': {n: p.detach().to('cpu', copy=True) for n, p in sel if 'lora_' in n},
               'extra_trainable': {n: p.detach().to('cpu', copy=True) for n, p in sel
                                   if 'lora_' not in n},
               'meta': dict(meta or {}, coverage=cov, projected_bytes=est)}
    tmp = path.with_name(path.name + '.tmp')
    torch.save(payload, tmp)
    actual = tmp.stat().st_size
    if actual > max_bytes:
        os.remove(tmp)
        raise RuntimeError('%s: wrote %d B, over budget %d B; removed.' % (path, actual, max_bytes))
    os.replace(tmp, path)
    return {'path': str(path), 'bytes': actual, 'coverage': cov}


def load_adapter_v3(model, path, map_location='cpu', allow_legacy=False):
    path = Path(path)
    payload = torch.load(path, map_location=map_location, weights_only=False)
    fmt = payload.get('format')
    named = dict(model.named_parameters())
    if fmt == V1_ADAPTER_FORMAT:
        if not allow_legacy:
            raise RuntimeError('%s: v1 adapter holds only lora_ keys (%d tensors). Any '

class Toy(nn.Module):
    def __init__(self, seed=0):
        super().__init__()
        torch.manual_seed(seed)
        self.dit = nn.Module()
        self.dit.base = nn.Linear(8, 8)
        self.dit.q_lora = nn.Module()
        self.dit.q_lora.lora_A = nn.Parameter(torch.randn(4, 8) * 0.1)
        self.dit.q_lora.lora_B = nn.Parameter(torch.randn(8, 4) * 0.1)
        self.dit.video_expert = nn.Linear(64, 64)
        self.proprio_encoder = nn.Linear(8, 64)
        self.dit.base.requires_grad_(False)

        '''Exactly trainer.py:85-88: LoRA by pattern, plus the proprio encoder.'''
        lora = [p for n, p in self.named_parameters() if 'lora_' in n]

def build(seed=0):
    m = Toy(seed)
    opt = torch.optim.AdamW(m.optimisable(), lr=1e-3)

        m.proprio_encoder(x).sum().backward()
        opt.step()
